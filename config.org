(defun ando-run-macro-until-line (line-number)
  "Run the last keyboard macro until LINE-NUMBER is reached."
  (interactive "nRun macro until line: ")
  (while (< (line-number-at-pos) line-number)
    (call-last-kbd-macro)))

(defun ando-yaml-indentation-level (s)
  (if (string-match "^ " s)
      (+ 1 (ando-yaml-indentation-level (substring s 1)))
    0))

(defun ando-yaml-current-line ()
  (buffer-substring-no-properties (line-beginning-position) (line-end-position)))

(defun ando-yaml-clean-string (s)
  (let* ((s (replace-regexp-in-string "^[[:blank:]-:]*" "" s))
         (s (replace-regexp-in-string ":$" "" s)))
    s))

(defun ando-yaml-not-blank-p (s)
  (string-match "[^[:blank:]]" s))

(defun ando-yaml-path-to-point ()
  (save-excursion
    (let* ((line (ando-yaml-current-line))
           (level (ando-yaml-indentation-level line))
           result)
      (while (> (point) (point-min))
        (beginning-of-line 0)
        (setq line (ando-yaml-current-line))
        (let ((new-level (ando-yaml-indentation-level line)))
          (when (and (ando-yaml-not-blank-p line)
                     (< new-level level))
            (setq level new-level)
            (setq result (push (ando-yaml-clean-string line) result)))))
      (mapconcat 'identity result " => "))))

(defun ando-yaml-show-path-to-point ()
  (interactive)
  (message (ando-yaml-path-to-point)))

(with-eval-after-load 'yaml-ts-mode
  (define-key yaml-ts-mode-map (kbd "C-c y p") #'ando-yaml-show-path-to-point))

(defun ando-wait-for-buffer (buffer-name)
  "Wait for the specified buffer to open."
  (while (not (get-buffer buffer-name))
    (sit-for 0.1)))

(defun ando-ruby-namespaced ()
  (interactive)
  (let ((buffer (get-buffer "*eldoc*")))
    (when buffer
      (kill-buffer buffer)
      (message "Buffer '%s' killed." "*eldoc*")))
  (eldoc--invoke-strategy t)
  (ando-wait-for-buffer "*eldoc*")
  (with-current-buffer "*eldoc*"
    (goto-char (point-min))
    (kill-new (buffer-substring (line-beginning-position) (line-end-position)))
    (kill-buffer "*eldoc*")))

(defun ando-copy-filename-to-clipboard ()
  "Copy the current buffer file name to the clipboard."
  (interactive)
  (let ((filename (if (equal major-mode 'dired-mode)
                      default-directory
                    (buffer-file-name))))
    (when filename
      (kill-new filename)
      (message "Copied buffer file name '%s' to the clipboard." filename))))

(defun ando-get-repo-name ()
  "Get magit current repo's name"
  (nth 0
       (last
        (split-string
         (magit-toplevel) "/") 2)))

(defun ando-file-path ()
  "Get file-path relative to repo"
  (string-join (nthcdr 5 (split-string buffer-file-name "/")) "/"))

(defun ando-file-in-ca-gh ()
  "Copy a file to its address in master on CA Github"
  (interactive)
  (let ((file-gh-address (format "https://github.com/cultureamp/%s/blob/master/%s#L%s"
                                 (ando-get-repo-name)
                                 (ando-file-path)
                                 (line-number-at-pos))))
    (kill-new file-gh-address)
    (message "Copied file's github address '%s' to the clipboard." file-gh-address)))

(defun ando-ticket-url ()
  "Open the jira ticket related to the current branch"
  (interactive)
  (let* ((jira-prefix "https://cultureamp.atlassian.net/browse/")
         (branch-ticket (nth 0 (split-string (magit-get-current-branch) "/"))))
    (format "%s%s" jira-prefix branch-ticket)))

(defun ando-insert-ticket-url ()
  "insert jira ticket related to the current branch"

  (insert (ando-ticket-url)))

(defun ando-open-ticket ()
  "Open the jira ticket related to the current branch"

    (open_in_browser ticket-url)

(defun ando-insert-formatted-date ()
  "Prompt for a date format and insert the resulting formatted date at point."
  (interactive)
  (let* ((formats '(("%Y%m%d" . "YYYYMMDD")
                    ("%Y%m%d%H%M%S" . "YYYYMMDDHHMMSS")
                    ("%Y-%m-%d" . "YYYY-MM-DD")
                    ("%Y-%m-%d %H:%M:%S" . "YYYY-MM-DD HH:MM:SS")
                    ("%d-%m-%Y" . "DD-MM-YYYY")
                    ("%d/%m/%Y" . "DD/MM/YYYY")
                    ("%H:%M:%S" . "HH:MM:SS")))
         (choices (mapcar #'cdr formats))
         (selection (completing-read "Choose date format: " choices))
         (format-string (car (rassoc selection formats)))
         (date (shell-command-to-string (concat "date +\"" format-string "\""))))
    (insert (string-trim date))))

(use-package copilot
  :straight (:host github :repo "copilot-emacs/copilot.el" :branch "main")
  :hook (prog-mode . copilot-mode)
  :bind (:map copilot-completion-map
              ("<tab>" . ando/copilot-or-corfu-tab)
              ("TAB" . ando/copilot-or-corfu-tab))
  :config
  (defun ando/copilot-or-corfu-tab ()
    "Accept copilot completion, or fall through to corfu if its popup is active."
    (interactive)
    (if (and (bound-and-true-p corfu-mode)
             (frame-live-p (bound-and-true-p corfu--frame))
             (frame-visible-p corfu--frame))
        (corfu-next 1)
      (copilot-accept-completion))))
